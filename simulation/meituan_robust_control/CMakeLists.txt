cmake_minimum_required(VERSION 3.8)
project(meituan_robust_control)
find_package(ament_cmake REQUIRED)
find_package(joint_trajectory_controller REQUIRED)
find_package(pluginlib REQUIRED)
find_package(kdl_parser REQUIRED)
find_package(orocos_kdl REQUIRED)
find_package(std_msgs REQUIRED)
add_library(robust_controller SHARED src/robust_controller.cpp)
target_include_directories(robust_controller PRIVATE include)
target_compile_features(robust_controller PUBLIC cxx_std_17)
ament_target_dependencies(robust_controller joint_trajectory_controller pluginlib kdl_parser orocos_kdl std_msgs)
pluginlib_export_plugin_description_file(controller_interface plugins.xml)
install(TARGETS robust_controller LIBRARY DESTINATION lib)
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_sliding test/test_sliding.cpp)
  target_include_directories(test_sliding PRIVATE include)
  target_compile_features(test_sliding PRIVATE cxx_std_17)
  add_test(NAME sliding_feedback COMMAND test_sliding)
endif()
ament_package()
